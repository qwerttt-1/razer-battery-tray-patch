cmake_minimum_required(VERSION 3.20)
project(RazerTrayPatchInstaller LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(PAYLOAD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/payload")
configure_file(src/resources.rc.in resources.rc @ONLY)
file(GLOB PATCH_ICONS "${PAYLOAD_DIR}/icons/*.ico")
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/resources.rc" PROPERTIES OBJECT_DEPENDS "${PAYLOAD_DIR}/patch.cjs;${PAYLOAD_DIR}/batteryTrayPatch.js;${PATCH_ICONS}")
add_executable(RazerTrayPatchInstaller WIN32 src/installer.cpp "${CMAKE_CURRENT_BINARY_DIR}/resources.rc")
target_compile_definitions(RazerTrayPatchInstaller PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(RazerTrayPatchInstaller PRIVATE /W4 /utf-8 /EHsc)
target_link_libraries(RazerTrayPatchInstaller PRIVATE comctl32 shell32 ole32 user32 gdi32 advapi32)
target_link_options(RazerTrayPatchInstaller PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
